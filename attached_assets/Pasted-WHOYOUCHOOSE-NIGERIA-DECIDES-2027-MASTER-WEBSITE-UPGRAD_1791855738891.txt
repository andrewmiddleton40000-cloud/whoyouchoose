function firstValue(row, keys, fallback = null) {  
  if (!row || typeof row !== "object") return fallback;  
  for (const key of keys) {  
    if (row[key] !== null && row[key] !== undefined && row[key] !== "") {  
      return row[key];  
    }  
  }  
  return fallback;  
}  
  
function getCandidateId(row) {  
  return firstValue(  
    row,  
    ["candidate_id", "id", "candidateId", "candidate_uuid"],  
    null  
  );  
}  
  
function getCandidateName(row) {  
  return String(  
    firstValue(row, ["candidate_name", "name", "full_name", "candidate"], "Unnamed candidate")  
  );  
}  
  
function getCandidateParty(row) {  
  return String(  
    firstValue(row, [  
      "party_name",  
      "party",  
      "party_acronym",  
      "party_abbreviation",  
      "party_name_abbreviation",  
      "party_code"  
    ], "Party not listed")  
  );  
}  
  
function getVotes(row) {  
  const value = Number(  
    firstValue(row, ["votes", "vote_count", "total_votes", "count", "votes_count"], 0)  
  );  
  return Number.isFinite(value) && value >= 0 ? value : 0;  
}  
  
function normalizeRows(data) {  
  if (Array.isArray(data)) return data;  
  if (data && Array.isArray(data.data)) return data.data;  
  if (data && Array.isArray(data.rows)) return data.rows;  
  if (data && Array.isArray(data.results)) return data.results;  
  return [];  
}  
  
function initials(name) {  
  return name.trim().split(/\s+/).slice(0, 2)  
    .map(part => part[0]).join("").toUpperCase() || "?";  
}  
  
function safeImage(url) {  
  try {  
    const parsed = new URL(url);  
    return parsed.protocol === "https:" || parsed.protocol === "http:";  
  } catch (_) {  
    return false;  
  }  
}

  if (fromRow && safeImage(String(fromRow))) return String(fromRow);
